const fs = require("fs");
const { execFileSync } = require("child_process");
const { chromium } = require("/opt/node22/lib/node_modules/playwright");
const MD = "/home/user/nairaplate/docs/NairaPlate_Master_Business_Rules_and_Audit_Specification.md";
let md = fs.readFileSync(MD, "utf8").replace(/^---[\s\S]*?---\n/, "");
// drop the markdown title block (the cover page shows it)
md = md.replace(/^# NairaPlate: Master Business Rules and Audit Specification\n\n\*\*Version 1.1, 3 October 2026\*\*\n\n/, "");
fs.writeFileSync("/tmp/master_body.md", md);
let html = execFileSync("python3", ["-c", "import markdown,sys; print(markdown.markdown(open('/tmp/master_body.md').read(), extensions=['tables','sane_lists']))"], { maxBuffer: 1 << 26 }).toString();
// wrap tables that have 6 or more columns in a landscape block
html = html.replace(/<table>([\s\S]*?)<\/table>/g, (m, inner) => {
  const head = inner.match(/<thead>([\s\S]*?)<\/thead>/);
  const cols = head ? (head[1].match(/<th/g) || []).length : 0;
  return (cols >= 6 || (head && head[1].includes("<th>Trigger</th>"))) ? `<div class="wide"><table>${inner}</table></div>` : `<table>${inner}</table>`;
});
// keep a section's heading and intro with its wide table, so there is no near-empty portrait page
html = html.replace(/(<h1>[^<]*<\/h1>\s*(?:<p>(?:(?!<\/p>)[\s\S])*<\/p>\s*)*)<div class="wide">/g, '<div class="wide">$1');
html = html.replace(/<table>(\s*<thead>\s*<tr>\s*<th>#<\/th>)/g, '<table class="num">$1');
// first-column emphasis
html = html.replace(/<tr>\s*<td>/g, '<tr><td class="first">');
const heads = [...html.matchAll(/<h1>([\s\S]*?)<\/h1>/g)].map((m) => m[1]);
const toc = `<h1 class="tochead">Contents</h1><ol class="toc">${heads.map((h) => `<li>${h}</li>`).join("")}</ol>`;

  @top-left { content: "NairaPlate"; font: 700 8.5pt Figtree; color: #0B1F33; vertical-align: bottom; padding-bottom: 3mm; border-bottom: 0.3mm solid #A6A6A6; }
  @bottom-left { content: "Version 1.1   |   3 October 2026"; font: 400 8.5pt Figtree; color: #0B1F33; border-top: 0.3mm solid #A6A6A6; padding-top: 3mm; vertical-align: top; }
  @bottom-right { content: "Page " counter(page); font: 400 8.5pt Figtree; color: #595959; border-top: 0.3mm solid #A6A6A6; padding-top: 3mm; vertical-align: top; }
}

.cover img { width: 62mm; margin-bottom: 14mm; } .title { font: 700 27pt/1.15 Figtree, Arial, sans-serif; color:#0B1F33; margin: 0 0 3mm; } .sub { color: #1677D2; font-size: 13pt; margin: 0 0 8mm; }

h1 { font: 700 15pt Figtree, Arial, sans-serif; color: #0B1F33; margin: 8mm 0 3mm; break-after: avoid; border-left: 1.4mm solid #1677D2; padding-left: 3mm; page-break-before: always; }

const cover = `<div class="cover"><img src="${logo}"><p class="title">Master Business Rules<br>and Audit Specification</p><p class="sub">How money, stock and records work in NairaPlate</p>

(async () => {
  const doc = `<!doctype html><html><head><meta charset="utf-8"><style>${css}</style></head><body>${cover}${toc}${html}</body></html>`;

  const b = await chromium.launch({ executablePath: "/opt/pw-browsers/chromium", args: ["--no-sandbox"] });

  await pg.goto("file://" + process.cwd() + "/master_preview.html", { waitUntil: "load" });

  await pg.pdf({ path: "/home/user/nairaplate/docs/NairaPlate_Master_Business_Rules_and_Audit_Specification.pdf", preferCSSPageSize: true, printBackground: true });
  await b.close();
  console.log("pdf written");
})().catch((e) => { console.error(e); process.exit(1); });
